# Packet structure:
# 8 bits: packet id
# 4 bits: packet type
# 1 bit: sender (0 = host, 1 = device)
# 8 bits: pin number (recommended: 0xFF when its not applicable, its ignored when not applicable anyway)
# N bits: data
#
# packet id is only supplied by the host, the device always responds using
# the same packet id. they are reused, check packet timings for debugging
# data length is connected to packet types and the sender
#
#
# Packet types:
#
# 0x0: echo (data length doesnt matter, it gets echoed back)
# 0x1: reserved (get device info)
# 0x2: ack (no data, sent from the device to indicate success)
# 0x3: reserved
# 0x4: set pin direction (1 bit data: 0 = input, 1 = output)
# 0x5: set internal pull-up (1 bit data: 0 = off, 1 = on)
# 0x6: reserved
# 0x7: reserved
# 0x8: write pin value (1 bit data: 0 = low, 1 = high)
# 0x9: get pin value (host: 0 bit. device: 1 bit: 0 = low, 1 = high)
# 0xA: listen to pin value changes (host: 1 bit: 0 = off, 1 = on. device: 1 bit: 0 = low, 1 = high)
# 0xB: reserved
# 0xC: reserved
# 0xD: reserved
# 0xE: reserved
# 0xF: reserved

from bitarray import bitarray

SENDER_HOST = 0
SENDER_DEVICE = 1

PIN_NA = 0xFF  # pin not applicable


class PacketType:
    ECHO = 0x0
    GET_INFO = 0x1  # reserved
    ACK = 0x2
    SET_DIRECTION = 0x4
    SET_PULLUP = 0x5
    WRITE_PIN = 0x8
    GET_PIN = 0x9
    LISTEN_PIN = 0xA
